% Nodes
\node (start) [startstop] {Start};
\node (in1) [io, below of=start]
{\verb|Input [3:0] a, [3:0] b, i = 0|};
\node (proc0) [process, below of=in1]
{\texttt{result = 0;}};

\node (proc1) [process, below of=proc0, align=left]
{

\verb|result = result ^ (a & mask);|
};

\node (dec1) [decision, below of=proc1]
{\verb|a[3] == 1|};

\node (proc2) [process, right of=dec1, xshift=4cm]
{\verb|a = a << 1;|};

\node (proc3) [process, below of=dec1]
{\verb|a = (a << 1) ^ 4'b0011;|};

\node (proc4) [process, below of=proc3]
{\verb|b = b >> 1;|};

\node (proc5) [process, below of=proc4]
{\verb|i++;|};

\node (dec2) [decision, below of=proc5]
{\verb|i < 4|};

\node (stop) [startstop, below of=dec2] {Stop};

\draw [arrow] (start) -- (in1);
\draw [arrow] (in1) -- (proc0);
\draw [arrow] (proc0) -- (proc1);

\draw [arrow] (dec1.east) -- node[above]{False} (proc2.west);

\draw [arrow] (proc2) |- (proc4);
\draw [arrow] (proc3) -- (proc4);

\draw [arrow] (proc4) -- (proc5);
\draw [arrow] (proc5) -- (dec2);

\draw [arrow] (dec2.south) -- node[right]{False} (stop.north);
\draw [arrow] (dec2.west) -- ++(-3,0)|- node[pos=0.25,above, xshift=-0.5cm]{True} (proc1.west);

\end{tikzpicture}
